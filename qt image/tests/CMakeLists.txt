cmake_minimum_required(VERSION 3.21)
project(LibraryUiSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
qt_standard_project_setup()
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_SOURCE_DIR}/..")
qt_add_executable(LibraryUiSmoke ui_smoke.cpp ../mainwindow.cpp ../libraryshell.cpp ../mainwindow.h ../mainwindow.ui)
target_include_directories(LibraryUiSmoke PRIVATE ..)
target_link_libraries(LibraryUiSmoke PRIVATE Qt6::Widgets)
